{{/* render-link.html Purpose: - Custom Markdown link render hook - Added to support glossary links (see glossary section) - Resolves relative links to: - pages - page bundle resources - global resources - Handles special cases: - "*" glossary shorthand - in-page anchors Originally added for glossary support. Do not remove unless glossary logic is migrated elsewhere. */}} {{- $u := urls.Parse .Destination -}} {{- $href := $u.String -}} {{- if eq .Text "*" -}} {{- $href = "" -}} {{- with or .Title (site.GetPage (printf "/glossary/%s" .Destination)).Title -}} {{ $.Destination }} {{- else -}} {{ .Destination }} {{- end -}} {{- else if strings.HasPrefix $u.String "#" -}} {{- $href = printf "%s#%s" .PageInner.RelPermalink $u.Fragment -}} {{- else if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($.PageInner.GetPage $path) ($.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $href = printf "%s?%s" $href . -}} {{- end -}} {{- with $u.Fragment -}} {{- $href = printf "%s#%s" $href . -}} {{- end -}} {{- end -}} {{- end -}} {{- if $href -}} {{- $attributes := dict "href" $href "title" (.Title | transform.HTMLEscape) -}} {{ .Text }} {{- end -}}
